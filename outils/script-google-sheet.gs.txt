/*
 * CAN & BOTTLE COLLEC — script a coller dans Extensions > Apps Script
 * de ton Google Sheet. Voir le README.md, section "Ajout automatise",
 * pour la procedure complete de deploiement.
 */
function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var sheet = findCollectionSheet();

    if (!sheet) {
      throw new Error("Aucun onglet avec une colonne 'Nom' n'a ete trouve.");
    }

    var lastColumn = sheet.getLastColumn();
    var headers = sheet.getRange(1, 1, 1, lastColumn).getValues()[0];

    var row = headers.map(function (header) {
      var value = data.row[header];
      return value !== undefined ? value : "";
    });

    sheet.appendRow(row);

    return ContentService
      .createTextOutput(JSON.stringify({ ok: true }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService
      .createTextOutput(JSON.stringify({ ok: false, error: String(err) }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

/* Cherche, parmi tous les onglets du classeur, celui dont la premiere
   ligne contient une colonne nommee "Nom" (peu importe sa position parmi
   les onglets). Evite de se tromper d'onglet si la collection n'est pas
   sur le tout premier. */
function findCollectionSheet() {
  var sheets = SpreadsheetApp.getActiveSpreadsheet().getSheets();
  for (var i = 0; i < sheets.length; i++) {
    var sheet = sheets[i];
    var lastColumn = sheet.getLastColumn();
    if (lastColumn === 0) continue;
    var headers = sheet.getRange(1, 1, 1, lastColumn).getValues()[0];
    if (headers.indexOf("Nom") !== -1) {
      return sheet;
    }
  }
  return null;
}
